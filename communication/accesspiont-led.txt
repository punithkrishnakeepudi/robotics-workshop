ESP8266WebServer server(80);

#define LED_PIN D0

int brightness = 0; // 0-255

let timeout = null;

function updateLedGlow(value){
    let glow = value / 4;
    const led = document.getElementById("led");
    led.style.opacity = Math.max(0.15, value/255);
    led.style.boxShadow =
        `0 0 ${glow}px #00ff00,
         0 0 ${glow*2}px #00ff00,
         0 0 ${glow*4}px #00ff00`;
}

function sendBrightness(value) {
    fetch(`/brightness?value=${value}`);
}

function sliderChanged(value){
    document.getElementById("value").innerHTML = value;
    updateLedGlow(value);

    // Debounce to prevent flooding the ESP8266 web server while sliding smoothly
    clearTimeout(timeout);
    timeout = setTimeout(() => {
        sendBrightness(value);
    }, 50);
}

function setBrightness(value){
    document.getElementById("brightnessSlider").value = value;
    document.getElementById("value").innerHTML = value;
    updateLedGlow(value);

void handleRoot() {
  server.send(200, "text/html", getHTML());
}

void handleBrightness() {
  if (server.hasArg("value")) {
    brightness = constrain(server.arg("value").toInt(), 0, 255);

    // For regular external LEDs (Active HIGH):
    // 255 -> Full brightness (ON)
    // 0 -> Zero brightness (OFF)
    analogWrite(LED_PIN, brightness);

    Serial.print("Brightness Sent: ");
    Serial.println(brightness);
  }

  server.send(200, "text/plain", "OK");
}

void setup() {
  Serial.begin(115200);

  pinMode(LED_PIN, OUTPUT);
  analogWriteRange(255);

  WiFi.mode(WIFI_AP);
  WiFi.softAP(ssid, password);

  Serial.println();
  Serial.print("AP IP: ");
  Serial.println(WiFi.softAPIP());

  server.on("/", handleRoot);
  server.on("/brightness", handleBrightness);

  server.begin();
  Serial.println("Server Started");
}

void loop() {
  server.handleClient();
}
