/*
   4-DOF ROBOTIC ARM - ESP8266 MQTT Controller

   Listens for gesture angles published by 7-robot-arm-media-pipe.py
   and drives four servos with the same smoothing as the web version.

  ---------------------------------------------------------------------
  WIRING (NodeMCU pin labels)
  ---------------------------------------------------------------------
      Base     -> D1 (GPIO5)
      Shoulder -> D2 (GPIO4)
      Elbow    -> D5 (GPIO14)
      Gripper  -> D6 (GPIO12)

  Servos need their own 5V supply (they pull far more than USB gives).
  Tie the supply ground to the NodeMCU ground or the signal is garbage.

  ---------------------------------------------------------------------
  MESSAGE FORMAT
  ---------------------------------------------------------------------
      "base,shoulder,elbow,gripper"      e.g.  "90,120,45,30"

  Library Manager: PubSubClient by Nick O'Leary
*/

#include <ESP8266WiFi.h>
#include <PubSubClient.h>
#include <Servo.h>

// Update these with values suitable for your network
const char* ssid     = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";

// Must match the MQTT broker and topic in the Python script
const char* mqtt_server = "broker.hivemq.com";
const int   mqtt_port   = 1883;
const char* mqtt_topic  = "grrr_workshop/esp8266/arm";

WiFiClient espClient;
PubSubClient client(espClient);

// =====================================================================
// JOINT TABLE  -  the single source of truth
// These limits must match LIMITS in the Python script.
// =====================================================================
#define NUM_JOINTS 4

enum { J_BASE = 0, J_SHOULDER, J_ELBOW, J_GRIPPER };

const uint8_t     JOINT_PIN[NUM_JOINTS]  = { D1,     D2,         D5,      D6        };
const char *const JOINT_NAME[NUM_JOINTS] = { "base", "shoulder", "elbow", "gripper" };
const int         JOINT_MIN[NUM_JOINTS]  = {   0,     20,         10,      20       };
const int         JOINT_MAX[NUM_JOINTS]  = { 180,    160,        170,     150       };
const int         JOINT_HOME[NUM_JOINTS] = {  90,     90,         90,      90       };

// 500-2400us gives SG90/MG90S their full travel; the library default clips it.
const int SERVO_MIN_US = 500;
const int SERVO_MAX_US = 2400;

Servo servos[NUM_JOINTS];
int   currentAngle[NUM_JOINTS];   // where the servo actually is
int   targetAngle[NUM_JOINTS];    // where the gesture wants it

// =====================================================================
// MOTION SMOOTHING
//   3 deg per 15 ms  ~=  200 deg/sec
//   This is what stops the arm snapping when your hand jumps.
// =====================================================================
const int           MAX_STEP_DEG     = 3;
const unsigned long STEP_INTERVAL_MS = 15;

unsigned long lastStepMs   = 0;
unsigned long lastMessageMs = 0;

// If the laptop dies or the hand tracking stops, park the arm instead of
// leaving it frozen mid-air under load.
const unsigned long FAILSAFE_MS = 4000;

void setTarget(int i, int angle) {
  targetAngle[i] = constrain(angle, JOINT_MIN[i], JOINT_MAX[i]);
}

void goHome() {
  for (int i = 0; i < NUM_JOINTS; i++) targetAngle[i] = JOINT_HOME[i];
}

// Called from loop(). Moves each joint a small step toward its target.
void updateServos() {
  unsigned long now = millis();
  if (now - lastStepMs < STEP_INTERVAL_MS) return;
  lastStepMs = now;

  for (int i = 0; i < NUM_JOINTS; i++) {
    if (currentAngle[i] == targetAngle[i]) continue;

    int delta = targetAngle[i] - currentAngle[i];
    if (delta >  MAX_STEP_DEG) delta =  MAX_STEP_DEG;
    if (delta < -MAX_STEP_DEG) delta = -MAX_STEP_DEG;

    currentAngle[i] += delta;
    servos[i].write(currentAngle[i]);
  }
}

void setup_wifi() {
  delay(10);
  Serial.println();
  Serial.print("Connecting to WiFi: ");
  Serial.println(ssid);

  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("");
  Serial.println("WiFi connected successfully!");
  Serial.print("IP address: ");
  Serial.println(WiFi.localIP());
}

// Splits "90,120,45,30" into four angles
void callback(char* topic, byte* payload, unsigned int length) {
  String message = "";
  for (unsigned int i = 0; i < length; i++) message += (char)payload[i];

  if (message == "HOME") {
    goHome();
    Serial.println("HOME");
    lastMessageMs = millis();
    return;
  }

  int joint = 0;
  int start = 0;

  // walk the string, cutting a number out at every comma
  for (int i = 0; i <= (int)message.length() && joint < NUM_JOINTS; i++) {
    if (i == (int)message.length() || message[i] == ',') {
      setTarget(joint, message.substring(start, i).toInt());
      joint++;
      start = i + 1;
    }
  }

  // a short message would leave stale joints, so ignore anything incomplete
  if (joint != NUM_JOINTS) {
    Serial.print("Bad message: ");
    Serial.println(message);
    return;
  }

  lastMessageMs = millis();

  for (int i = 0; i < NUM_JOINTS; i++) {
    Serial.print(JOINT_NAME[i]); Serial.print("="); Serial.print(targetAngle[i]);
    Serial.print(i < NUM_JOINTS - 1 ? "  " : "\n");
  }
}

void reconnect() {
  while (!client.connected()) {
    Serial.print("Attempting MQTT connection...");

    String clientId = "ESP8266Arm-";
    clientId += String(random(0xffff), HEX);

    if (client.connect(clientId.c_str())) {
      Serial.println("connected");
      client.subscribe(mqtt_topic);
    } else {
      Serial.print("failed, rc=");
      Serial.print(client.state());
      Serial.println(" try again in 5 seconds");

      // keep easing the servos while we wait, do not block for 5s straight
      unsigned long wait = millis();
      while (millis() - wait < 5000) { updateServos(); yield(); }
    }
  }
}

void setup() {
  Serial.begin(115200);
  delay(300);

  for (int i = 0; i < NUM_JOINTS; i++) {
    currentAngle[i] = JOINT_HOME[i];
    targetAngle[i]  = JOINT_HOME[i];
    servos[i].attach(JOINT_PIN[i], SERVO_MIN_US, SERVO_MAX_US);
    servos[i].write(currentAngle[i]);
  }

  Serial.println();
  Serial.println(F("=============================="));
  Serial.println(F("  4-DOF ARM - GESTURE CONTROL"));
  Serial.println(F("=============================="));

  setup_wifi();

  client.setServer(mqtt_server, mqtt_port);
  client.setCallback(callback);

  lastMessageMs = millis();
}

void loop() {
  if (!client.connected()) reconnect();
  client.loop();

  // nothing heard for a while? park it rather than hold a pose under load
  if (millis() - lastMessageMs > FAILSAFE_MS) {
    goHome();
    lastMessageMs = millis();
  }

  updateServos();   // non-blocking easing toward each target
  yield();
}
