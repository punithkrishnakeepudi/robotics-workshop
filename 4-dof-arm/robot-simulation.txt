/*
   4-DOF ROBOTIC ARM — ESP8266 Access Point Controller  (optimized)

  ---------------------------------------------------------------------
  WIRING (NodeMCU pin labels)
  ---------------------------------------------------------------------
      Base     -> D1 (GPIO5)
      Shoulder -> D2 (GPIO4)
      Elbow    -> D5 (GPIO14)
      Gripper  -> D6 (GPIO12)

*/

// =====================================================================

ESP8266WebServer server(80);

// =====================================================================
// JOINT TABLE  —  the single source of truth
// Edit pins and limits HERE only; the web page reads them at load.
// =====================================================================
#define NUM_JOINTS 4

enum { J_BASE = 0, J_SHOULDER, J_ELBOW, J_GRIPPER };

const uint8_t     JOINT_PIN[NUM_JOINTS]  = { D1,     D2,         D5,      D6        };
const char *const JOINT_NAME[NUM_JOINTS] = { "base", "shoulder", "elbow", "gripper" };
const int         JOINT_MIN[NUM_JOINTS]  = {   0,     20,         10,      20       };
const int         JOINT_MAX[NUM_JOINTS]  = { 180,    160,        170,     150       };
const int         JOINT_HOME[NUM_JOINTS] = {  90,     90,         90,      90       };

// Servo pulse range. 500-2400us gives SG90/MG90S their full travel;
// the library default (544-2400) clips a few degrees at the low end.
const int SERVO_MIN_US = 500;
const int SERVO_MAX_US = 2400;

Servo servos[NUM_JOINTS];
int   currentAngle[NUM_JOINTS];   // where the servo actually is
int   targetAngle[NUM_JOINTS];    // where the slider wants it

// =====================================================================
// MOTION SMOOTHING
//   MAX_STEP_DEG / STEP_INTERVAL_MS = speed.
//   3 deg per 15 ms  ~=  200 deg/sec  (smooth, low current spikes)
//   Raise MAX_STEP_DEG for a snappier arm, lower it for a gentler one.
// =====================================================================
const int           MAX_STEP_DEG     = 3;
const unsigned long STEP_INTERVAL_MS = 15;

unsigned long lastStepMs = 0;
bool servosAttached = false;

// =====================================================================
// SERVO HELPERS
// =====================================================================
void attachServos() {
  if (servosAttached) return;
  for (int i = 0; i < NUM_JOINTS; i++) {
    servos[i].attach(JOINT_PIN[i], SERVO_MIN_US, SERVO_MAX_US);
    servos[i].write(currentAngle[i]);
  }
  servosAttached = true;
}

void detachServos() {
  if (!servosAttached) return;
  for (int i = 0; i < NUM_JOINTS; i++) servos[i].detach();
  servosAttached = false;
}

// Called from loop(). Moves each joint a small step toward its target.
void updateServos() {
  unsigned long now = millis();
  if (now - lastStepMs < STEP_INTERVAL_MS) return;
  lastStepMs = now;

  if (!servosAttached) return;

  for (int i = 0; i < NUM_JOINTS; i++) {
    if (currentAngle[i] == targetAngle[i]) continue;

    int delta = targetAngle[i] - currentAngle[i];
    if (delta >  MAX_STEP_DEG) delta =  MAX_STEP_DEG;
    if (delta < -MAX_STEP_DEG) delta = -MAX_STEP_DEG;

    currentAngle[i] += delta;
    servos[i].write(currentAngle[i]);
  }
}

// Returns joint index for a name, or -1
int jointIndex(const String &name) {
  for (int i = 0; i < NUM_JOINTS; i++) {
    if (name.equals(JOINT_NAME[i])) return i;
  }
  return -1;
}

void setTarget(int i, int angle) {
  targetAngle[i] = constrain(angle, JOINT_MIN[i], JOINT_MAX[i]);
}

function setStatus(cls, text){

/* -------------------------------------------------------------
   REQUEST QUEUE
   The ESP8266 serves one connection at a time. Never have more
   than one request open; keep only the newest value per joint
   and send it the moment the previous request completes.
------------------------------------------------------------- */
var inFlight = false;
var pending  = {};        // joint name -> newest unsent angle
var failures = 0;

function pump(){
  if (inFlight) return;
  var names = Object.keys(pending);
  if (!names.length) return;

  var name  = names[0];
  var value = pending[name];
  delete pending[name];

  inFlight = true;
  fetch("/servo?name=" + name + "&angle=" + value)
    .then(function(){
      if (failures) { failures = 0; setStatus("", "Connected"); }
    })
    .catch(function(){
      if (++failures > 2) setStatus("lost", "Reconnecting");
    })
    .then(function(){ inFlight = false; pump(); });
}

function send(name, value){
  pending[name] = value;
  pump();
}

/* Send all four joints in ONE request (used by Home). */
function sendAll(values){
  var q = JOINTS.map(function(n){ return n + "=" + values[n]; }).join("&");
  fetch("/all?" + q).catch(function(){});
}

  ctx.fillStyle = "#262626"; ctx.fillRect(baseX-45, baseY-18, 90, 20);

    ctx.stroke();
  });
}

// =====================================================================
// HTTP HANDLERS
// =====================================================================
void handleRoot() {
  server.send_P(200, "text/html", MAIN_PAGE);
}

// /servo?name=elbow&angle=120
void handleServo() {
  if (!server.hasArg("name") || !server.hasArg("angle")) {
    server.send(400, "text/plain", "Missing parameters");
    return;
  }

  int i = jointIndex(server.arg("name"));
  if (i < 0) {
    server.send(400, "text/plain", "Unknown joint");
    return;
  }

  attachServos();                            // undo a previous Relax
  setTarget(i, server.arg("angle").toInt());
  server.send(200, "text/plain", "OK");
}

// /all?base=90&shoulder=90&elbow=90&gripper=90   (one round trip)
void handleAll() {
  attachServos();
  for (int i = 0; i < NUM_JOINTS; i++) {
    if (server.hasArg(JOINT_NAME[i])) {
      setTarget(i, server.arg(JOINT_NAME[i]).toInt());
    }
  }
  server.send(200, "text/plain", "OK");
}

// Current angle, commanded target and limits for every joint
void handleState() {
  char json[320];
  int n = 0;
  n += snprintf(json + n, sizeof(json) - n, "{");
  for (int i = 0; i < NUM_JOINTS; i++) {
    n += snprintf(json + n, sizeof(json) - n,
                  "%s\"%s\":{\"current\":%d,\"target\":%d,\"min\":%d,\"max\":%d}",
                  (i ? "," : ""), JOINT_NAME[i],
                  currentAngle[i], targetAngle[i], JOINT_MIN[i], JOINT_MAX[i]);
  }
  n += snprintf(json + n, sizeof(json) - n, ",\"attached\":%s}",
                servosAttached ? "true" : "false");
  server.send(200, "application/json", json);
}

// Cut drive power so the servos stop humming/holding.
// NOTE: a heavy arm may sag under gravity when relaxed.
// The next slider move re-attaches them automatically.
void handleRelax() {
  detachServos();
  server.send(200, "text/plain", "RELAXED");
}

void handleNotFound() {
  server.send(404, "text/plain", "Not found");
}

// =====================================================================
// SETUP
// =====================================================================
void setup() {
  Serial.begin(115200);
  delay(300);

  for (int i = 0; i < NUM_JOINTS; i++) {
    currentAngle[i] = JOINT_HOME[i];
    targetAngle[i]  = JOINT_HOME[i];
  }
  attachServos();

  WiFi.persistent(false);      // don't wear out flash rewriting Wi-Fi config
  WiFi.mode(WIFI_AP);
  WiFi.softAP(AP_SSID, AP_PASSWORD, 1 /*channel*/, false /*hidden*/, 4 /*max clients*/);

  Serial.println();
  Serial.println(F("=============================="));
  Serial.println(F("  4-DOF ROBOT ARM"));
  Serial.println(F("=============================="));
  Serial.print(F("WiFi SSID : ")); Serial.println(AP_SSID);
  Serial.print(F("Password  : ")); Serial.println(AP_PASSWORD);
  Serial.print(F("Open URL  : http://")); Serial.println(WiFi.softAPIP());
  Serial.println(F("=============================="));

  server.on("/",       HTTP_GET, handleRoot);
  server.on("/servo",  HTTP_GET, handleServo);
  server.on("/all",    HTTP_GET, handleAll);
  server.on("/state",  HTTP_GET, handleState);
  server.on("/relax",  HTTP_GET, handleRelax);
  server.onNotFound(handleNotFound);
  server.begin();

  Serial.println(F("Web server started."));
}

// =====================================================================
// LOOP
// =====================================================================
void loop() {
  server.handleClient();
  updateServos();      // non-blocking easing toward each target
  yield();
}
